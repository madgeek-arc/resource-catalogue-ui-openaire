# The scheme the client used: TLS ends before this server, so $scheme alone is always http.
map $http_x_forwarded_proto $forwarded_proto {
    default $http_x_forwarded_proto;
    ""      $scheme;
}

server {
    ${ADD_SERVER_NAME}
    listen 80;
    root /usr/share/nginx/html;

    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/css text/javascript application/javascript application/json image/svg+xml;

    location / {
        try_files $uri$args $uri$args/ /index.html;
    }

    # Angular names its bundles <name>-<8 character hash>.js|css (main-WYIZ6B7F.js, chunk-H7rKH-Sx.js), so a changed file always
    # gets a new URL and these can be cached for a year. index.html, ngsw.json and the service worker files
    # carry no hash and are not matched.
    location ~ "-[A-Za-z0-9_-]{8}\.(js|css)$" {
        expires 1y;
    }

    location ~* \.(eot|ttf|woff)$ {
        add_header Access-Control-Allow-Origin *;
    }

    ${ADD_PROXY_API}

    location ~ ^/(images|files|ebooks|pitch-decks)/ {
        root /usr/share/nginx/html/data;
        autoindex on;
    }

}
